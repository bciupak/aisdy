### Przykładowa instrukcja - kompresja:
java -jar huffman.jar -m comp -s niemanie.txt -d comp.txt -l 1
java -jar huffman.jar -m comp -s imie.txt -d comp2.txt -l 1

### Przykładowa instrukcja - dekompresja:
java -jar huffman.jar -m decomp -s comp.txt -d decomp.txt -l 1
java -jar huffman.jar -m decomp -s comp2.txt -d decomp2.txt -l 1
